# 蟊星者事件
# msi_dsai_events：maoxingzhe.1 蚕食宗主国（每 5 年），maoxingzhe.10 借壳复活。
# 触发器定义在 common/scripted_triggers/msi_dsai_triggers.txt。
# 钩子定义在 common/on_actions/msi_dsai_on_actions.txt。

namespace = maoxingzhe

###################################
# 蚕食宗主国（每 5 年 1 个星球）
###################################
country_event = {
	id = maoxingzhe.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_valid_civic = civic_maoxingzhe
		is_subject = yes
		exists = overlord
		overlord = {
			is_country_type = default
			NOT = { is_maoxingzhe_immune = yes }
			# 注意：这里不排除首都。没有任何相邻星系时，要把首都也吃掉。
			any_owned_planet = {
				is_colony = yes
			}
		}
	}

	immediate = {
		# 优先吞并“与蟊星者领地相邻”的宗主国星系
		if = {
			limit = {
				overlord = {
					any_owned_planet = {
						is_colony = yes
						solar_system = {
							any_neighbor_system = {
								any_system_planet = {
									has_owner = yes
									owner = { is_same_value = root }
								}
							}
						}
					}
				}
			}
			overlord = {
				random_owned_planet = {
					limit = {
						is_colony = yes
						solar_system = {
							any_neighbor_system = {
								any_system_planet = {
									has_owner = yes
									owner = { is_same_value = root }
								}
							}
						}
					}
					set_owner = root
					set_planet_flag = maoxingzhe_devoured
				}
			}
		}
		# 没有任何相邻星系时，直接吞并宗主国首都
		else = {
			overlord = {
				capital_scope = {
					set_owner = root
					set_planet_flag = maoxingzhe_devoured
				}
			}
		}
	}
}

###################################
# 借壳复活
###################################
country_event = {
	id = maoxingzhe.10
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_valid_civic = civic_maoxingzhe
		exists = from
		from = {
			is_country_type = default
			NOT = { is_maoxingzhe_immune = yes }
			any_owned_planet = {
				is_colony = yes
				is_capital = no
			}
		}
	}

	immediate = {
		# 先在战胜国境内挑一块落脚地（不动首都）
		from = {
			random_owned_planet = {
				limit = {
					is_colony = yes
					is_capital = no
				}
				save_event_target_as = maoxingzhe_rebirth_planet
			}
		}

		if = {
			limit = { exists = event_target:maoxingzhe_rebirth_planet }
			event_target:maoxingzhe_rebirth_planet = {
				create_rebels = {
					name = random
					authority = auth_dictatorial
					civics = {
						civic = civic_maoxingzhe
						civic = random
						civic = random
					}
					species = root
					ethos = {
						ethic = ethic_fanatic_authoritarian
						ethic = ethic_xenophobe
					}
					flag = random
					effect = {
						set_country_flag = maoxingzhe_reborn
					}
				}
			}
		}
	}
}

# 战争主理人（civic_war_broker）：
# 与本国关系中立以上的国家进入战争且本国未参战时，给参战国 +33% 舰船射速与陆军伤害。
# 触发源：on_war_beginning（Root = 参战国, From = War）
# scope: this = 参战国
country_event = {
	id = msi_dsai.6
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_at_war = yes
	}

	immediate = {
		every_country = {
			limit = {
				has_valid_civic = civic_war_broker
				is_at_war = no
				# is_same_value 只接受 = <target> 或 = { target = <target> }，
				# 没有 value 参数；要表达“不是 root 自己”须外面套 NOT。
				NOT = { is_same_value = root }
				NOR = {
					is_hostile = root
					is_rival = root
				}
				opinion = {
					who = root
					value >= 0
				}
			}
			root = {
				add_modifier = {
					modifier = msi_dsai_war_broker_boost
					days = -1
				}
			}
		}
	}
}

# 战争主理人：战争结束后移除第三方赠予的 boost

country_event = {

	hide_window = yes

	immediate = {
		remove_modifier = msi_dsai_war_broker_boost
	}
}

# 天人感应（civic_heaven_human_resonance）：
# 每年检查本国殖民地，犯罪率 > 60 或稳定度 < 60 时，在该星球所在星系生成灵能风暴。
# 触发源：on_yearly_pulse_country
# scope: this = country
country_event = {
	id = msi_dsai.8
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_valid_civic = civic_heaven_human_resonance
		has_cosmic_storms_dlc = yes
	}

	immediate = {
		every_owned_colony = {
			limit = {
				OR = {
					planet_crime > 60
					planet_stability < 60
				}
			}
			solar_system = {
				spawn_random_storm = {
					type = shroud_storm
					cosmic_storm_start_position = this
					immediate = yes
				}
			}
		}
	}
}

# 毅种循环（civic_evolutionary_cycle）：
# 社会竞争每隔两年物理淘汰 1% 的人口。
# 触发源：on_yearly_pulse_country（每年触发），用国家 flag 交替实现“每两年执行一次”。
# 淘汰方式：遍历全国每个 pop group，各 2% 概率杀掉该群组中的 1 个 pop。
# scope: this = country
country_event = {
	id = msi_dsai.9
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_valid_civic = civic_evolutionary_cycle
	}

	immediate = {
		if = {
			limit = {
				has_country_flag = msi_dsai_evolution_cycle_tick
			}
			remove_country_flag = msi_dsai_evolution_cycle_tick
			every_owned_colony = {
				every_owned_pop_group = {
					random_list = {
						90 = {}
						10 = { kill_single_pop = yes }
					}
				}
			}
		}
		else = {
			set_country_flag = msi_dsai_evolution_cycle_tick
		}
	}
}

# 古老习俗（civic_ancient_customs）：
# 每年 1 月 1 日，宗主国送来一份资源红包，并附上一句随机的新年祝福。
# 红包按本国殖民地数量逐颗发放，规模随帝国成长。
# 祝福语取自网络上长辈对晚辈的新年祝福（12 句，见 localisation/<lang>/msi_dsai_event_newyear_l_*.yml）。
# msi_dsai.10 负责判定与抽祝福（隐藏），msi_dsai.11 负责弹窗（格式参考原版先辈子弟问候事件 origin.10）。
# 触发源：on_yearly_pulse_country
# scope: this = country
country_event = {
	id = msi_dsai.10
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_valid_civic = civic_ancient_customs
		is_subject = yes
		exists = overlord
		# overlord must be a fallen empire or awakened fallen empire
		overlord = {
			OR = {
				is_country_type = fallen_empire
				is_country_type = awakened_fallen_empire
			}
		}
	}

	immediate = {
		# 先清掉去年的祝福标记，保证每年只有一条祝福生效
		remove_country_flag = msi_dsai_newyear_1
		remove_country_flag = msi_dsai_newyear_2
		remove_country_flag = msi_dsai_newyear_3
		remove_country_flag = msi_dsai_newyear_4
		remove_country_flag = msi_dsai_newyear_5
		remove_country_flag = msi_dsai_newyear_6
		remove_country_flag = msi_dsai_newyear_7
		remove_country_flag = msi_dsai_newyear_8
		remove_country_flag = msi_dsai_newyear_9
		remove_country_flag = msi_dsai_newyear_10
		remove_country_flag = msi_dsai_newyear_11
		remove_country_flag = msi_dsai_newyear_12
		# 宗主随机挑一句新年祝福
		random_list = {
			1 = { set_country_flag = msi_dsai_newyear_1 }
			1 = { set_country_flag = msi_dsai_newyear_2 }
			1 = { set_country_flag = msi_dsai_newyear_3 }
			1 = { set_country_flag = msi_dsai_newyear_4 }
			1 = { set_country_flag = msi_dsai_newyear_5 }
			1 = { set_country_flag = msi_dsai_newyear_6 }
			1 = { set_country_flag = msi_dsai_newyear_7 }
			1 = { set_country_flag = msi_dsai_newyear_8 }
			1 = { set_country_flag = msi_dsai_newyear_9 }
			1 = { set_country_flag = msi_dsai_newyear_10 }
			1 = { set_country_flag = msi_dsai_newyear_11 }
			1 = { set_country_flag = msi_dsai_newyear_12 }
		}
		country_event = { id = msi_dsai.11 days = 0 }
	}
}
