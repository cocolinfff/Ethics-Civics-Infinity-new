# Hidden setup event, fired from on_game_start_country.
# It only schedules the discovery event; the dig site itself is created by lilith_mia.2.
# Only human players are allowed to receive the site.
country_event = {
	id = lilith_mia.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ai = no
		exists = capital_scope
		has_global_flag = eci_special_leader_yes
	}

	immediate = {
		country_event = { id = lilith_mia.2 days = 20 }
	}
}

# Discovery event: this is where the archaeological site is actually created.
country_event = {
	id = lilith_mia.2
	title = "lilith_mia.2.name"
	desc = "lilith_mia.2.desc"
	picture = GFX_evt_glitchy_matrix

	is_triggered_only = yes

	trigger = {
		is_ai = no
		exists = capital_scope
		NOT = { has_country_flag = lilith_mia_discovery_shown }
	}

	immediate = {
		set_country_flag = lilith_mia_discovery_shown
		lilith_mia_spawn_digsite_effect = yes
		if = {
			limit = {
				lilith_mia_digsite_exists_trigger = yes
			}
			set_country_flag = lilith_mia_digsite_spawned
		}
		else = {
			# Extremely rare safety net: try again a little later.
			country_event = { id = lilith_mia.4 days = 120 }
		}
	}

	option = {
		name = lilith_mia.2.a
	}
}

# Safety net retry. Does nothing if the site was created by lilith_mia.2.
country_event = {
	id = lilith_mia.4
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ai = no
		exists = capital_scope
		NOT = { has_country_flag = lilith_mia_digsite_spawned }
	}

	immediate = {
		lilith_mia_spawn_digsite_effect = yes
		if = {
			limit = {
				lilith_mia_digsite_exists_trigger = yes
			}
			set_country_flag = lilith_mia_digsite_spawned
		}
	}
}

# Step 2: the legendary leader recruitment window.
country_event = {
	id = lilith_mia.15
	title = "lilith_mia.15.name"
	desc = "lilith_mia.15.desc"
	picture = GFX_leader_recruitment_bg_renowned
	event_window_type = leader_recruit

	is_triggered_only = yes

	trigger = {
		exists = event_target:lilith_mia_leader
	}

	picture_event_data = {
		portrait = event_target:lilith_mia_leader
	}

	option = {
		name = HIRE
		tag = hire_leader
		hidden_effect = {
			clone_leader = {
				target = event_target:lilith_mia_leader
				tier = leader_tier_legendary
			}
			last_created_leader = {
				save_global_event_target_as = lilith_mia_hired
				set_leader_flag = immune_to_negative_traits
				set_leader_flag = leader_death_events_blocked
			}
			create_message = {
				type = MESSAGE_RECRUITED_LEADER
				localization = MESSAGE_RECRUITED_LEADER_DESC
				days = 10
				target = last_created_leader
				variable = {
					type = name
					localization = LEADER
					scope = last_created_leader
				}
				variable = {
					key = "border"
					value = "GFX_border_legendary"
				}
			}
		}
	}

	option = {
		name = lilith_mia.15.d
		tag = dismiss_leader
		custom_tooltip = lilith_mia.15.d.custom_tooltip
		hidden_effect = {
			clone_leader = {
				target = event_target:lilith_mia_leader
				tier = leader_tier_legendary
			}
			last_created_leader = {
				save_global_event_target_as = lilith_mia_hired
				set_leader_flag = immune_to_negative_traits
				set_leader_flag = leader_death_events_blocked
			}
			create_message = {
				type = MESSAGE_RECRUITED_LEADER
				localization = MESSAGE_RECRUITED_LEADER_DESC
				days = 10
				target = last_created_leader
				variable = {
					type = name
					localization = LEADER
					scope = last_created_leader
				}
				variable = {
					key = "border"
					value = "GFX_border_legendary"
				}
			}
		}
	}
