# 力工思维（Force Labor Mind）
# 【宗教信仰】平民在繁重劳动中放弃深度思考，转而怨恨那些引导他们思考的人。
#
# 需求：
#   是竞争主义或精英主义  → 竞争主义 = ethic_authoritarian，精英主义 = ethic_fanatic_authoritarian
#                          两档但**不含**中立思潮调和主义，故用含 any_ 的键
#   是精神主义或教权主义  → 精神主义 = ethic_spiritualist，教权主义 = ethic_fanatic_spiritualist
#                          两档但**不含**中立思潮实用主义，同样用含 any_ 的键
#
# 两条需求是 AND：写在同一个 ethics 块里的两个 OR 就是 AND 关系
# （与上面的蟊星者同构，也与原版政体 README 的措辞一致）。
#
# 效果实现说明：
#   civic 只能挂国家级固定修正（modifier 块），无法按“某个思潮的人口的数量”做逐条判定，
#   因此这里用与需求语义最贴近的固定值：
#     planet_stability_add                  —— 精英主义劳工平民带来的秩序
#     pop_cat_worker_happiness              —— 平等主义氛围下劳工的不满
#     pop_cat_civilian_happiness            —— 同上，平民
#     pop_ethic_egalitarian_attraction_mult —— 劳苦大众越是没有话语权，平等主义越没人信
civic_force_labor_mind = {
	description = civic_tooltip_force_labor_mind_effects

	potential = {
		ethics = {
			NOT = {
				value = ethic_gestalt_consciousness
			}
		}
	}

	possible = {
		ethics = {
			# 是竞争主义或精英主义（不含中立思潮调和主义）
			OR = {
				text = civic_tooltip_any_authoritarian
				value = ethic_authoritarian
				value = ethic_fanatic_authoritarian
			}
			# 是精神主义或教权主义（不含中立思潮实用主义）
			OR = {
				text = civic_tooltip_any_spiritualist
				value = ethic_spiritualist
				value = ethic_fanatic_spiritualist
			}
		}
	}

	random_weight = {
		base = 10
	}

	# 效果不在 civic 上挂固定修正，而是拆到具体载体：
	#   第一条（精英主义劳工/平民提高稳定度）→ common/pop_categories/正常人口阶级.txt
	#     的 worker / civilian 类别里，按人口组自身的思潮判定
	#   第二、第三条（平等主义劳工平民的不满、平等主义吸引力衰减）
	#     → common/deposits/ethic_deposit.txt 的 d_ethic_society 里
}
