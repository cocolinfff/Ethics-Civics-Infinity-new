# Scripted triggers shared by the two paragon event chains.

# True if the Su dig site exists in the capital system. Scope: country.
msi_dsai_su_digsite_exists_trigger = {
	capital_scope? = {
		solar_system = {
			any_system_planet = {
				has_planet_flag = msi_dsai_su_digsite_planet
			}
		}
	}
}

# True if the AM-001 anomaly has been planted in the capital system. Scope: country.
msi_dsai_am001_anomaly_exists_trigger = {
	capital_scope? = {
		solar_system = {
			any_system_planet = {
				has_anomaly = yes
				has_planet_flag = msi_dsai_am001_anomaly_planet
			}
		}
	}
}
