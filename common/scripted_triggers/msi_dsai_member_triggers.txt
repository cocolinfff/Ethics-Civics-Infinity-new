# ============================================================================
# 成员国管理 —— 共用触发器
# 系统只对拥有 civic_coalition_government 或 civic_coalition_government_overlord
# 的文明开启；星际邦联（星际联邦）与联合政府共用同一套成员国数据。
# ============================================================================

# 是否拥有可启用成员国系统的两个理念之一
msi_dsai_ms_has_access = {
	is_valid_coalition_government = yes
}

# 系统是否已初始化（初始化时会为首都挂上全部五个成员国的主权修正）
msi_dsai_ms_is_initialized = {
	has_country_flag = msi_dsai_ms_initialized
}

# 当前选中的成员国是否为轮值主席（轮值主席不可被制裁）
# 用显式五组比较，避免 check_variable 右值变量在不同版本的解析差异。
msi_dsai_ms_selected_is_chair = {
	OR = {
		AND = {
			check_variable = { which = msi_dsai_ms_selected value = 1 }
			check_variable = { which = msi_dsai_ms_chair value = 1 }
		}
		AND = {
			check_variable = { which = msi_dsai_ms_selected value = 2 }
			check_variable = { which = msi_dsai_ms_chair value = 2 }
		}
		AND = {
			check_variable = { which = msi_dsai_ms_selected value = 3 }
			check_variable = { which = msi_dsai_ms_chair value = 3 }
		}
		AND = {
			check_variable = { which = msi_dsai_ms_selected value = 4 }
			check_variable = { which = msi_dsai_ms_chair value = 4 }
		}
		AND = {
			check_variable = { which = msi_dsai_ms_selected value = 5 }
			check_variable = { which = msi_dsai_ms_chair value = 5 }
		}
		AND = {
			check_variable = { which = msi_dsai_ms_selected value = 6 }
			check_variable = { which = msi_dsai_ms_chair value = 6 }
		}
		AND = {
			check_variable = { which = msi_dsai_ms_selected value = 7 }
			check_variable = { which = msi_dsai_ms_chair value = 7 }
		}
		AND = {
			check_variable = { which = msi_dsai_ms_selected value = 8 }
			check_variable = { which = msi_dsai_ms_chair value = 8 }
		}
	}
}

# 当前选中的成员国编号 >= 1 且 <= 5（面板与按钮的前置校验）
msi_dsai_ms_selected_valid = {
	check_variable = { which = msi_dsai_ms_selected value >= 1 }
	check_variable = { which = msi_dsai_ms_selected value <= 8 }
}

# ---------------------------------------------------------------------------
# 起源“和平共处”（origin_advanced_start_independent）与首都割据地块
#
# 若玩家起源为和平共处，且首都（母星）上存在下列割据地块，则对应的成员国
# 在联合政府面板中不可被选中与操作。仅限 country scope 使用（capital_scope）。
# ---------------------------------------------------------------------------
msi_dsai_ms_blocked_1 = {
	has_origin = origin_advanced_start_independent
	capital_scope = { has_deposit = d_eci_United_States }
}

msi_dsai_ms_blocked_3 = {
	has_origin = origin_advanced_start_independent
	capital_scope = { has_deposit = d_eci_Wasteland_Federation }
}

msi_dsai_ms_blocked_4 = {
	has_origin = origin_advanced_start_independent
	capital_scope = { has_deposit = d_eci_United_Kingdom }
}

msi_dsai_ms_blocked_5 = {
	has_origin = origin_advanced_start_independent
	capital_scope = { has_deposit = d_eci_Consortium }

msi_dsai_ms_blocked_6 = {
	has_origin = origin_advanced_start_independent
	capital_scope = { has_deposit = d_eci_Warlord_Nations }
}

# 当前选中的成员国是否处于上述不可操作状态
msi_dsai_ms_selected_blocked = {
	OR = {
		AND = {
			check_variable = { which = msi_dsai_ms_selected value = 1 }
			msi_dsai_ms_blocked_1 = yes
		}
		AND = {
			check_variable = { which = msi_dsai_ms_selected value = 3 }
			msi_dsai_ms_blocked_3 = yes
		}
		AND = {
			check_variable = { which = msi_dsai_ms_selected value = 4 }
			msi_dsai_ms_blocked_4 = yes
		}
		AND = {
			check_variable = { which = msi_dsai_ms_selected value = 5 }
			msi_dsai_ms_blocked_5 = yes
		}
		AND = {
			check_variable = { which = msi_dsai_ms_selected value = 6 }
			msi_dsai_ms_blocked_6 = yes
		}
	}
}

# 当前轮值主席国是否因首都割据地块而不可担任（用于轮换时跳过）
