# ============================================================================
# 成员国管理 —— 面板按钮效果
#
# 面板窗口（interface/msi_dsai_member_panel.gui）里的全部可交互按钮都走这里。
# 注意：effectbuttonType 的 This 是“当前选中的对象”（行星/舰队/系统等），
# 没有选中对象时才是玩家国家；From 为玩家国家。因此每个按钮都同时处理
# country 与 planet 两种 scope，任何一类上下文下都不会失效。
# ============================================================================

# ---------------------------------------------------------------------------
# 侧边栏：选择成员国（只改当前选中项并重开面板，不产生其他效果）
# ---------------------------------------------------------------------------
msi_dsai_ms_btn_pick_1 = {
	potential = { always = yes }
	allow = { always = yes }
	effect = {
		custom_tooltip = msi_dsai_ms_tip_pick
		hidden_effect = {
			if = {
				limit = { is_scope_type = country }
				set_variable = { which = msi_dsai_ms_selected value = 1 }
				country_event = { id = msi_ms.10 }
			}
			else = {
				owner = {
					set_variable = { which = msi_dsai_ms_selected value = 1 }
					country_event = { id = msi_ms.10 }
				}
			}
		}
	}
}

# ---------------------------------------------------------------------------
# 操作一：拉拢成员国（消耗 200 影响力，满意度 +15，5 年修正）
# ---------------------------------------------------------------------------
msi_dsai_ms_btn_woo = {
	potential = { always = yes }
	allow = {
		custom_tooltip = {
			fail_text = msi_dsai_ms_fail_blocked
			hidden_trigger = {
				OR = {
					AND = {
						is_scope_type = country
						NOT = { msi_dsai_ms_selected_blocked = yes }
					}
					AND = {
						is_scope_type = planet
						owner = { NOT = { msi_dsai_ms_selected_blocked = yes } }
					}
				}
			}
		}
		custom_tooltip = {
			fail_text = msi_dsai_ms_fail_influence_200
			hidden_trigger = {
				OR = {
					AND = {
						is_scope_type = country
						resource_stockpile_compare = { resource = influence value >= 200 }
					}
					AND = {
						is_scope_type = planet
						owner = { resource_stockpile_compare = { resource = influence value >= 200 } }
					}
				}
			}
		}
	}
	effect = {
		custom_tooltip = msi_dsai_ms_tip_woo
		hidden_effect = {
			if = {
				limit = { is_scope_type = country }
				msi_dsai_ms_woo_apply = yes
			}
			else = {
				owner = { msi_dsai_ms_woo_apply = yes }
			}
		}
	}
}

# ---------------------------------------------------------------------------
# 操作二：移交星球主权
# 选中了行星时直接把该行星交给当前选中的成员国；
# 没有选中行星时给出操作指引（行星决议里也能完成同样的移交）。
# ---------------------------------------------------------------------------
msi_dsai_ms_btn_transfer = {
	potential = { always = yes }
	allow = {
		custom_tooltip = {
			fail_text = msi_dsai_ms_fail_blocked
			hidden_trigger = {
				OR = {
					AND = {
						is_scope_type = country
						NOT = { msi_dsai_ms_selected_blocked = yes }
					}
					AND = {
						is_scope_type = planet
						owner = { NOT = { msi_dsai_ms_selected_blocked = yes } }
					}
				}
			}
		}
		custom_tooltip = {
			fail_text = msi_dsai_ms_fail_capital
			hidden_trigger = {
				OR = {
					is_scope_type = country
					AND = {
						is_scope_type = planet
						is_capital = no
					}
				}
			}
		}
	}
	effect = {
		custom_tooltip = msi_dsai_ms_tip_transfer
		hidden_effect = {
			if = {
				# 面板从行星决议打开时，按钮的作用域就是那颗星球；
				# 首都不参与主权授予，一律走指引分支。
				limit = { is_scope_type = planet is_capital = no }
				msi_dsai_ms_planet_assign_selected = yes
				owner = { country_event = { id = msi_ms.10 } }
			}
			else = {
				country_event = { id = msi_ms.22 }
			}
		}
	}
}
