# 删除星球上所有 d_ethic_private_commercial_t2N 地块
# 逻辑：星球上有多少个该地块，就 remove 多少个
# while 循环直到该星球上此类地块数量为 0 为止（参照原版 remove_all_default_rural_districts 的写法）
remove_building_private_commercial_megaplex = {
    while = {
        limit = {
            count_deposits = {
                type = d_ethic_private_commercial_t2N
                count > 0
            }
        }
        remove_deposit = d_ethic_private_commercial_t2N
    }
}
