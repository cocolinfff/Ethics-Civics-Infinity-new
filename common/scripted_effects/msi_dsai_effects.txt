# 谬误平衡 —— 动态档位刷新
# 由 common/on_actions/msi_dsai_on_actions.txt 的 on_yearly_pulse_country 调用。
# Scope: country
#
# 逻辑：统计内阁（councilor）所有负面特质的层级总数，按总数挂载对应档位的
# 岗位效率修正（每档 +5%，见 common/static_modifiers/msi_dsai_modifiers.txt）。
# 每年刷新一次，先摘后挂，保证降级与失效都能正确回收。

msi_dsai_wupingheng_refresh = {
	remove_modifier = msi_dsai_wupingheng_bonus_1
	remove_modifier = msi_dsai_wupingheng_bonus_2
	remove_modifier = msi_dsai_wupingheng_bonus_3
	remove_modifier = msi_dsai_wupingheng_bonus_4
	remove_modifier = msi_dsai_wupingheng_bonus_5

	if = {
		limit = { has_valid_civic = civic_mius_wupingheng }

		set_variable = { which = msi_dsai_wupingheng_count value = 0 }

		every_owned_leader = {
			limit = { is_councilor = yes }
			change_variable = {
				which = msi_dsai_wupingheng_count
				value = value:leader_num_negative_leader_trait_tiers
			}
		}

		if = {
			limit = {
				check_variable = { which = msi_dsai_wupingheng_count value >= 5 }
			}
			add_modifier = { modifier = msi_dsai_wupingheng_bonus_5 }
		}
		else_if = {
			limit = {
				check_variable = { which = msi_dsai_wupingheng_count value >= 4 }
			}
			add_modifier = { modifier = msi_dsai_wupingheng_bonus_4 }
		}
		else_if = {
			limit = {
				check_variable = { which = msi_dsai_wupingheng_count value >= 3 }
			}
			add_modifier = { modifier = msi_dsai_wupingheng_bonus_3 }
		}
		else_if = {
			limit = {
				check_variable = { which = msi_dsai_wupingheng_count value >= 2 }
			}
			add_modifier = { modifier = msi_dsai_wupingheng_bonus_2 }
		}
		else_if = {
			limit = {
				check_variable = { which = msi_dsai_wupingheng_count value >= 1 }
			}
			add_modifier = { modifier = msi_dsai_wupingheng_bonus_1 }
		}

		clear_variable = msi_dsai_wupingheng_count
	}
}
