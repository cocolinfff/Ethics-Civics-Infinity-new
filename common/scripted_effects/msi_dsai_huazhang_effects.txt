# scope: country（玩家）
huazhang_create_wangfeng_leader = {
	create_country = {
		name = "NAME_Unknown"
		type = global_event
		auto_delete = yes
		flag = {
			icon = {
				category = "red_dawn"
				file = "red_dawn.dds"
			}
			background = {
				category = "backgrounds"
				file = "00_solid.dds"
			}
			colors = {
				"red"
				"red"
				"null"
				"null"
			}
		}
		effect = {
			save_event_target_as = wangfeng_display_country
		}
	}
	event_target:wangfeng_display_country = {
		create_species = {
			name = "NAME_red_dawn"
			class = PARAGON
			portrait = wangfeng
			namelist = "HUMAN1"
			traits = {
				trait = trait_pc_nuked_preference

				trait = trait_psionic
				trait = trait_erudite
				trait = trait_cyborg_logic_engines
				trait = trait_cyborg_enhanced_memory
				trait = trait_cyborg_learning_algorithms
			}
			effect = {
				save_event_target_as = wangfeng_species
			}
		}
		create_leader = {
			class = commander
			tier = leader_tier_legendary
			species = event_target:wangfeng_species
			name = "NAME_Wangfeng"
			skill = 5
			gender = female
			event_leader = no
			skip_background_generation = yes
			randomize_traits = no
			custom_description = wangfeng1
			custom_catch_phrase = wangfeng1_catch_phrase
			background_ethic = ethic_egalitarian
			hide_age = yes
			traits = {
				trait = subclass_commander_admiral
				trait = leader_trait_red_dawn
				trait = leader_trait_wangfeng_fleet
				trait = leader_trait_wangfeng_ideology
				trait = leader_trait_wangfeng_government
				trait = leader_trait_wangfeng_produce
			}
			effect = {
				change_leader_portrait = wangfeng
				set_age = 17
				set_leader_flag = legendary_leader
				set_leader_flag = immune_to_negative_traits
				set_leader_flag = wangfeng
				exile_leader_as = wangfeng_display_leader
				save_global_event_target_as = renowned_wangfeng
			}
		}
	}
}

# 华章送给玩家的“古代尖端技术战舰”。
# scope: 玩家首都（planet）
red_dowm_gift = {
	create_fleet = {
		name = "NAME_Laurel_Fleet"
		effect = {
			set_owner = event_target:red_dawn_enemy
			while = {
				count = 1
				create_fallen_empire_titan = yes
			}
			while = {
				count = 3
				create_fallen_empire_large_ship = yes
			}
			while = {
				count = 7
				create_fallen_empire_small_ship = yes
			}
			set_location = {
				target = event_target:huazhang_rest_area.capital_star
				distance = 45
				angle = random
			}
			set_owner = event_target:huazhang_rest_area
		}
	}
}

# 新赤色黎明向三个堕落帝国宣战。
# scope: event_target:red_dawn_enemy（新赤色黎明）
red_dowm_war = {
	every_country = {
		limit = {
			is_fallen_empire_spiritualist = yes
		}
		save_global_event_target_as = decayed_fallen_empire_1
	}
	every_country = {
		limit = {
			is_fallen_empire_xenophile = yes
		}
		save_global_event_target_as = decayed_fallen_empire_2
	}
	every_country = {
		limit = {
			is_fallen_empire_xenophobe = yes
		}
		save_global_event_target_as = decayed_fallen_empire_3
	}
	declare_war = {
		target = event_target:decayed_fallen_empire_1
		attacker_war_goal = wg_galactic_revolution
		name = "cb_galactic_revolution"
