# ============================================================================
# 成员国管理 —— 共用效果
#
# 数据模型（均为国家变量，5 个成员国编号 1~5）：
#   msi_dsai_ms_selected        当前面板选中的成员国
#   msi_dsai_ms_sat_<i>         满意度 0~100
#   msi_dsai_ms_ethic_<i>       思潮编号 1~8（见 msi_dsai_member_events.txt 的思潮选项）
#   msi_dsai_ms_planets_<i>     该国拥有的殖民星球数（由本文件重算）
#   msi_dsai_ms_chair           轮值主席国
#   msi_dsai_ms_chair_timer     轮值任期计时（年）
# 行星变量：
#   msi_dsai_ms_owner           0 = 联合政府直辖，1~5 = 对应成员国
# ============================================================================

# ---------------------------------------------------------------------------
# 初始化：只执行一次；把首都标记为五国共有，并写入全部初始数据。
# 由 msi_ms.41（开局）、msi_ms.40（改革后）、面板事件 msi_ms.10 共同调用。
# ---------------------------------------------------------------------------
msi_dsai_ms_init = {
	if = {
		limit = { NOT = { msi_dsai_ms_is_initialized = yes } }
		set_country_flag = msi_dsai_ms_initialized

		set_variable = { which = msi_dsai_ms_selected value = 1 }

		set_variable = { which = msi_dsai_ms_sat_1 value = @msi_dsai_ms_sat_start }
		set_variable = { which = msi_dsai_ms_sat_2 value = @msi_dsai_ms_sat_start }
		set_variable = { which = msi_dsai_ms_sat_3 value = @msi_dsai_ms_sat_start }
		set_variable = { which = msi_dsai_ms_sat_4 value = @msi_dsai_ms_sat_start }
		set_variable = { which = msi_dsai_ms_sat_5 value = @msi_dsai_ms_sat_start }

		# 初始思潮（编号见 msi_dsai_member_events.txt 的思潮选项）：
		# 合众联邦 多元主义(7) / 人民阵线 合作主义(3) / 钢铁国度 干涉主义(1)
		# 联合帝国 竞争主义(4) / 贸易同盟 实用主义(10)
		set_variable = { which = msi_dsai_ms_ethic_1 value = 7 }
		set_variable = { which = msi_dsai_ms_ethic_2 value = 3 }
		set_variable = { which = msi_dsai_ms_ethic_3 value = 1 }
		set_variable = { which = msi_dsai_ms_ethic_4 value = 4 }
		set_variable = { which = msi_dsai_ms_ethic_5 value = 10 }

		set_variable = { which = msi_dsai_ms_planets_1 value = 0 }
		set_variable = { which = msi_dsai_ms_planets_2 value = 0 }
		set_variable = { which = msi_dsai_ms_planets_3 value = 0 }
		set_variable = { which = msi_dsai_ms_planets_4 value = 0 }
		set_variable = { which = msi_dsai_ms_planets_5 value = 0 }

		# 和平共处起源：母星割据格局下由人民阵线担任轮值主席
		if = {
			limit = { has_origin = origin_advanced_start_independent }
			set_variable = { which = msi_dsai_ms_chair value = 2 }
		}
		else = {
			set_variable = { which = msi_dsai_ms_chair value = 1 }
		}
		set_variable = { which = msi_dsai_ms_chair_timer value = 0 }

		# 母星（首都）拥有全部五个成员国的主权修正
		if = {
			limit = { exists = capital_scope }
			capital_scope = {
				add_modifier = { modifier = msi_dsai_pm_ms_member_1 }
				add_modifier = { modifier = msi_dsai_pm_ms_member_2 }
				add_modifier = { modifier = msi_dsai_pm_ms_member_3 }
				add_modifier = { modifier = msi_dsai_pm_ms_member_4 }
				add_modifier = { modifier = msi_dsai_pm_ms_member_5 }
				set_variable = { which = msi_dsai_ms_owner value = 0 }
			}
		}

		msi_dsai_ms_recount_colonies = yes
		msi_dsai_ms_sat_sync = yes
	}
}

# ---------------------------------------------------------------------------
# 重算各成员国的殖民星球数（按行星上的 msi_dsai_ms_owner 变量统计）
# ---------------------------------------------------------------------------
msi_dsai_ms_recount_colonies = {
	set_variable = { which = msi_dsai_ms_planets_1 value = 0 }
	set_variable = { which = msi_dsai_ms_planets_2 value = 0 }
	set_variable = { which = msi_dsai_ms_planets_3 value = 0 }
	set_variable = { which = msi_dsai_ms_planets_4 value = 0 }
	set_variable = { which = msi_dsai_ms_planets_5 value = 0 }

	every_owned_planet = {
		limit = {
			is_colony = yes
			check_variable = { which = msi_dsai_ms_owner value = 1 }
		}
		owner = { change_variable = { which = msi_dsai_ms_planets_1 value = 1 } }
	}
	every_owned_planet = {
		limit = {
			is_colony = yes
			check_variable = { which = msi_dsai_ms_owner value = 2 }
		}
		owner = { change_variable = { which = msi_dsai_ms_planets_2 value = 1 } }
	}
	every_owned_planet = {
		limit = {
			is_colony = yes
			check_variable = { which = msi_dsai_ms_owner value = 3 }
		}
		owner = { change_variable = { which = msi_dsai_ms_planets_3 value = 1 } }
	}
	every_owned_planet = {
		limit = {
			is_colony = yes
			check_variable = { which = msi_dsai_ms_owner value = 4 }
		}
		owner = { change_variable = { which = msi_dsai_ms_planets_4 value = 1 } }
	}
	every_owned_planet = {
		limit = {
			is_colony = yes
			check_variable = { which = msi_dsai_ms_owner value = 5 }
		}
		owner = { change_variable = { which = msi_dsai_ms_planets_5 value = 1 } }
	}
}

# ---------------------------------------------------------------------------
# 满意度加减（参数：VAR = 满意度变量名，VALUE = 变动值）
# 结果夹在 0~100 之间。
# ---------------------------------------------------------------------------
msi_dsai_ms_sat_change = {
	change_variable = { which = $VAR$ value = $VALUE$ }
	if = {
		limit = { check_variable = { which = $VAR$ value > 100 } }
		set_variable = { which = $VAR$ value = 100 }
	}
	if = {
		limit = { check_variable = { which = $VAR$ value < 0 } }
		set_variable = { which = $VAR$ value = 0 }
	}
}

# 行星移交：把本行星授予指定成员国。
# 由行星决议、殖民弹窗与面板按钮调用（行星 scope）。
# 若该行星原本属于其他成员国，原成员国满意度大幅下降。
# ---------------------------------------------------------------------------

# 授予成员国 1：合众联邦
msi_dsai_ms_planet_assign_1 = {
	# 旧成员国不满
	if = {
		limit = { check_variable = { which = msi_dsai_ms_owner value = 2 } }
		owner = { msi_dsai_ms_sat_change = { VAR = msi_dsai_ms_sat_2 VALUE = @msi_dsai_ms_sat_takeover_penalty } }
	}
	if = {
		limit = { check_variable = { which = msi_dsai_ms_owner value = 3 } }
		owner = { msi_dsai_ms_sat_change = { VAR = msi_dsai_ms_sat_3 VALUE = @msi_dsai_ms_sat_takeover_penalty } }
	}
	if = {
		limit = { check_variable = { which = msi_dsai_ms_owner value = 4 } }
		owner = { msi_dsai_ms_sat_change = { VAR = msi_dsai_ms_sat_4 VALUE = @msi_dsai_ms_sat_takeover_penalty } }
	}
	if = {
		limit = { check_variable = { which = msi_dsai_ms_owner value = 5 } }
		owner = { msi_dsai_ms_sat_change = { VAR = msi_dsai_ms_sat_5 VALUE = @msi_dsai_ms_sat_takeover_penalty } }
	}

	# 新成员国满意（直辖地改为移交时加成更高）
	if = {
		limit = { check_variable = { which = msi_dsai_ms_owner value = 0 } }
		owner = { msi_dsai_ms_sat_change = { VAR = msi_dsai_ms_sat_1 VALUE = @msi_dsai_ms_sat_transfer } }
	}
	else = {
		owner = { msi_dsai_ms_sat_change = { VAR = msi_dsai_ms_sat_1 VALUE = 5 } }
	}

	remove_modifier = msi_dsai_pm_ms_member_1
	remove_modifier = msi_dsai_pm_ms_member_2
	remove_modifier = msi_dsai_pm_ms_member_3
	remove_modifier = msi_dsai_pm_ms_member_4
	remove_modifier = msi_dsai_pm_ms_member_5
	add_modifier = { modifier = msi_dsai_pm_ms_member_1 }
	set_variable = { which = msi_dsai_ms_owner value = 1 }

	owner = {
		msi_dsai_ms_recount_colonies = yes
		msi_dsai_ms_sat_sync = yes
	}
}

# 轮值主席跳过：若当前主席国被首都割据地块封锁，则顺延一位
# ---------------------------------------------------------------------------
msi_dsai_ms_chair_skip = {
	if = {
		limit = { msi_dsai_ms_chair_blocked = yes }
		change_variable = { which = msi_dsai_ms_chair value = 1 }
		if = {
			limit = { check_variable = { which = msi_dsai_ms_chair value > 5 } }
			set_variable = { which = msi_dsai_ms_chair value = 1 }
		}
	}

# ---------------------------------------------------------------------------
# 轮值主席推进：每年 +1，满 10 年顺延到下一国（1→2→…→5→1）
# 顺延后跳过被首都割据地块封锁的成员国（最多检查 5 次，覆盖连续封锁）
# ---------------------------------------------------------------------------
msi_dsai_ms_chair_advance = {
	change_variable = { which = msi_dsai_ms_chair_timer value = 1 }
	if = {
		limit = { check_variable = { which = msi_dsai_ms_chair_timer value >= @msi_dsai_ms_chair_term } }
		set_variable = { which = msi_dsai_ms_chair_timer value = 0 }
		change_variable = { which = msi_dsai_ms_chair value = 1 }
		if = {
			limit = { check_variable = { which = msi_dsai_ms_chair value > 5 } }
			set_variable = { which = msi_dsai_ms_chair value = 1 }
		}
		msi_dsai_ms_chair_skip = yes
		msi_dsai_ms_chair_skip = yes
		msi_dsai_ms_chair_skip = yes
		msi_dsai_ms_chair_skip = yes
		msi_dsai_ms_chair_skip = yes
	}
}

# ---------------------------------------------------------------------------
# 年度检查：殖民地差距惩罚 + 轮值主席的声望加成 + 重算 + 同步
# ---------------------------------------------------------------------------
msi_dsai_ms_yearly_check = {
	msi_dsai_ms_recount_colonies = yes
	msi_dsai_ms_chair_advance = yes

	# 各国"落后门槛" = 本国殖民地数 + 3
	set_variable = { which = msi_dsai_ms_gap_1 value = msi_dsai_ms_planets_1 }
	change_variable = { which = msi_dsai_ms_gap_1 value = @msi_dsai_ms_colony_gap }
	set_variable = { which = msi_dsai_ms_gap_2 value = msi_dsai_ms_planets_2 }
	change_variable = { which = msi_dsai_ms_gap_2 value = @msi_dsai_ms_colony_gap }
	set_variable = { which = msi_dsai_ms_gap_3 value = msi_dsai_ms_planets_3 }
	change_variable = { which = msi_dsai_ms_gap_3 value = @msi_dsai_ms_colony_gap }
	set_variable = { which = msi_dsai_ms_gap_4 value = msi_dsai_ms_planets_4 }
	change_variable = { which = msi_dsai_ms_gap_4 value = @msi_dsai_ms_colony_gap }
	set_variable = { which = msi_dsai_ms_gap_5 value = msi_dsai_ms_planets_5 }
	change_variable = { which = msi_dsai_ms_gap_5 value = @msi_dsai_ms_colony_gap }

	# 有任何一个成员国比该国多 3 个及以上殖民地时，满意度逐渐下降
	if = {
		limit = {
			OR = {
				check_variable = { which = msi_dsai_ms_planets_2 value >= msi_dsai_ms_gap_1 }
				check_variable = { which = msi_dsai_ms_planets_3 value >= msi_dsai_ms_gap_1 }
				check_variable = { which = msi_dsai_ms_planets_4 value >= msi_dsai_ms_gap_1 }
				check_variable = { which = msi_dsai_ms_planets_5 value >= msi_dsai_ms_gap_1 }
			}
		}
		msi_dsai_ms_sat_change = { VAR = msi_dsai_ms_sat_1 VALUE = @msi_dsai_ms_sat_gap_penalty }
	}
	if = {
		limit = {
			OR = {
				check_variable = { which = msi_dsai_ms_planets_1 value >= msi_dsai_ms_gap_2 }
				check_variable = { which = msi_dsai_ms_planets_3 value >= msi_dsai_ms_gap_2 }
				check_variable = { which = msi_dsai_ms_planets_4 value >= msi_dsai_ms_gap_2 }
				check_variable = { which = msi_dsai_ms_planets_5 value >= msi_dsai_ms_gap_2 }
			}
		}
		msi_dsai_ms_sat_change = { VAR = msi_dsai_ms_sat_2 VALUE = @msi_dsai_ms_sat_gap_penalty }
	}
	if = {
		limit = {
			OR = {
				check_variable = { which = msi_dsai_ms_planets_1 value >= msi_dsai_ms_gap_3 }
				check_variable = { which = msi_dsai_ms_planets_2 value >= msi_dsai_ms_gap_3 }
				check_variable = { which = msi_dsai_ms_planets_4 value >= msi_dsai_ms_gap_3 }
				check_variable = { which = msi_dsai_ms_planets_5 value >= msi_dsai_ms_gap_3 }
			}
		}
		msi_dsai_ms_sat_change = { VAR = msi_dsai_ms_sat_3 VALUE = @msi_dsai_ms_sat_gap_penalty }
	}
	if = {
		limit = {
			OR = {
				check_variable = { which = msi_dsai_ms_planets_1 value >= msi_dsai_ms_gap_4 }
				check_variable = { which = msi_dsai_ms_planets_2 value >= msi_dsai_ms_gap_4 }
				check_variable = { which = msi_dsai_ms_planets_3 value >= msi_dsai_ms_gap_4 }
				check_variable = { which = msi_dsai_ms_planets_5 value >= msi_dsai_ms_gap_4 }
			}
		}
		msi_dsai_ms_sat_change = { VAR = msi_dsai_ms_sat_4 VALUE = @msi_dsai_ms_sat_gap_penalty }
	}
	if = {
		limit = {
			OR = {
				check_variable = { which = msi_dsai_ms_planets_1 value >= msi_dsai_ms_gap_5 }
				check_variable = { which = msi_dsai_ms_planets_2 value >= msi_dsai_ms_gap_5 }
				check_variable = { which = msi_dsai_ms_planets_3 value >= msi_dsai_ms_gap_5 }
				check_variable = { which = msi_dsai_ms_planets_4 value >= msi_dsai_ms_gap_5 }
			}
		}
		msi_dsai_ms_sat_change = { VAR = msi_dsai_ms_sat_5 VALUE = @msi_dsai_ms_sat_gap_penalty }
	}

	# 轮值主席国获得声望加成
	if = {
		limit = { check_variable = { which = msi_dsai_ms_chair value = 1 } }
		msi_dsai_ms_sat_change = { VAR = msi_dsai_ms_sat_1 VALUE = @msi_dsai_ms_sat_chair_bonus }
	}
	if = {
		limit = { check_variable = { which = msi_dsai_ms_chair value = 2 } }
		msi_dsai_ms_sat_change = { VAR = msi_dsai_ms_sat_2 VALUE = @msi_dsai_ms_sat_chair_bonus }
	}
	if = {
		limit = { check_variable = { which = msi_dsai_ms_chair value = 3 } }
		msi_dsai_ms_sat_change = { VAR = msi_dsai_ms_sat_3 VALUE = @msi_dsai_ms_sat_chair_bonus }
	}
	if = {
		limit = { check_variable = { which = msi_dsai_ms_chair value = 4 } }
		msi_dsai_ms_sat_change = { VAR = msi_dsai_ms_sat_4 VALUE = @msi_dsai_ms_sat_chair_bonus }
	}
	if = {
		limit = { check_variable = { which = msi_dsai_ms_chair value = 5 } }
		msi_dsai_ms_sat_change = { VAR = msi_dsai_ms_sat_5 VALUE = @msi_dsai_ms_sat_chair_bonus }
	}

	msi_dsai_ms_sat_sync = yes
}
