# Lilith Mia - archaeological site
# The site is spawned by event lilith_mia.1 in the player's capital system.
# It is never added to the random pool (weight = 0).

lilith_mia_digsite = {
	desc = lilith_mia_digsite_desc
	picture = GFX_evt_sapient_AI
	stages = 4
	weight = 0
	max_instances = 999

	allow = {
		exists = leader
		leader = { leader_class = scientist }
		# Only human players may excavate this site, even if an AI empire captures the system.
		owner = { is_ai = no }
	}

	potential = {
		exists = leader
		leader = { leader_class = scientist }
	}

	visible = {
		is_ai = no
		default_site_visible_trigger = yes
	}

	on_create = {
		planet = {
			set_planet_flag = lilith_mia_digsite_planet
		}
	}

	stage = {
		difficulty = 5
		icon = GFX_archaeology_runes_A1
		event = lilith_mia.10
	}

	stage = {
		difficulty = 6
		icon = GFX_archaeology_runes_A2
		event = lilith_mia.11
	}

	stage = {
		difficulty = 7
		icon = GFX_archaeology_runes_A3
		event = lilith_mia.12
	}

	stage = {
		difficulty = 8
		icon = GFX_archaeology_runes_E2
		event = lilith_mia.13
	}

	on_roll_failed = {
		from = {
			standard_archaeological_site_on_roll_failed = { RANDOM_EVENTS = no_events }
		}
	}
}
