# Paragon traits for the two community-requested leaders.
# AM-001 (scientist) is a pre-FTL probe intelligence; Su (official) is a
# cryo-preserved administrator from the previous universe.
# Both traits are only handed out by their event chains, never rolled randomly.

leader_trait_eci_am001 = {
	force_councilor_trait = yes
	immortal_leaders = yes
	inline_script = {
		script = trait/icon
		CLASS = leader
		ICON = "GFX_leader_trait_machine_intelligence"
		RARITY = paragon
		COUNCIL = yes
		TIER = none
	}
	# The probe was built for research and its new core is still doing research.
	councilor_modifier = {
		all_technology_research_speed = 0.15
	}
	leader_class = { scientist }
	initial = no
	randomized = no
	hide_age = yes
}

leader_trait_eci_su = {
	force_councilor_trait = yes
	immortal_leaders = yes
	inline_script = {
		script = trait/icon
		CLASS = leader
		ICON = "GFX_leader_trait_ancient_knowledge"
		RARITY = paragon
		COUNCIL = yes
		TIER = none
	}
	# Obedience without doubt makes a colony run like a machine. It also makes
	# the people living there unhappy, which is exactly the trade-off intended.
	planet_modifier = {
		planet_jobs_produces_mult = 0.2
		pop_bonus_workforce_mult = 0.2
		planet_stability_add = 10
		planet_crime_add = -30
		pop_happiness = -0.1
	}
	councilor_modifier = {
		country_admin_cap_add = 30
		country_unity_produces_mult = 0.05
	}
	leader_class = { official }
	initial = no
	randomized = no
	hide_age = yes
}
