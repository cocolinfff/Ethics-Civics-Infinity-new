# ============================================================================
# 成员国管理 —— on_action 挂接
#
# 本模组的 on_action 为合并语义：同名定义会追加事件而非覆盖，
# 因此这里只追加，不会影响原版或其他文件的同类 on_action。
# ============================================================================

# 开局静默初始化（拥有两个理念之一的玩家国家）
# scope: this = country
on_game_start_country = {
	events = {
		msi_ms.41
	}
}

# 改革后：若新政府带有两个理念之一且尚未初始化，弹窗补挂首都主权修正；
#           若新政府不再带有两个理念，清理整套系统。
# scope: this = country
on_post_government_changed = {
	events = {
		msi_ms.40
		msi_ms.43
	}
}

# 新殖民完成：询问把星球授予哪个成员国
# scope: this = planet（Carrier）
on_colonized = {
	events = {
		msi_ms.20
	}
}

# 行星易主：新主人未启用成员国系统时清除主权标记
# scope: this = planet, from = new owner
on_colony_transfer = {
	events = {
		msi_ms.44
		msi_ms.45
	}
}

# 年度检查：殖民地差距、轮值主席轮换、满意度修正同步
# scope: this = country
on_yearly_pulse_country = {
	events = {
		msi_ms.50
	}
}
