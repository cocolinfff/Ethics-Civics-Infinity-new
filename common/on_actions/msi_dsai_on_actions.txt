# 蟊星者钩子
# 模组的 on_action 是合并语义（同名定义会追加事件而非覆盖），
# 因此这里只追加，不会影响原版或其他文件的同类 on_action。

# 每 5 年对一个国家触发一次 —— 用于“蚕食宗主国星球”
# scope: this = country
on_five_year_pulse_country = {
	events = {
		maoxingzhe.1
	}
}

# 国家被摧毁时触发 —— 用于“借壳复活”
# scope: this = destroyed country, from = optional destroyer (country)
on_country_destroyed = {
	events = {
		maoxingzhe.10
	}
}

# 每年触发一次 —— 用于“谬误平衡”按内阁负面特质数刷新岗位效率档位
#                  以及“开源盛世”按情报国数量刷新研究加成
#                  以及“毅种循环”隔年物理淘汰（由事件内部用 flag 交替成两年一次）
# scope: this = country
on_yearly_pulse_country = {
	events = {
		msi_dsai.1
		msi_dsai.3
		msi_dsai.8
		msi_dsai.9
	}
}

# 开局时对每个国家触发一次 —— 用于“高天之上”给母星放置两座太空电梯
# scope: this = country
on_game_start_country = {
	events = {
		msi_dsai.2
	}
}

# 造神运动：领袖升级 / 被雇佣 / 生成时，检查是否带负面特质
# scope: this = country, from = leader
on_leader_level_up = {
	events = {
		msi_dsai.4
	}
}

on_leader_hired = {
	events = {
		msi_dsai.4
	}
}

on_leader_spawned = {
	events = {
		msi_dsai.4
	}
}

# 战争主理人：战争开始时，给参战国挂上第三方赠予的 boost
# Root = 参战国, From = War
on_war_beginning = {
	events = {
		msi_dsai.6
	}
}

# 战争主理人：战争结束时移除 boost
on_war_ended = {
	events = {
		msi_dsai.7
	}
}
