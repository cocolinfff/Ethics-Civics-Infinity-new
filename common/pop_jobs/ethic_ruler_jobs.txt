merchant_msi = {
	category = ruler
	can_set_priority = no

	swappable_data = {
		default = {
			condition_string = RULER_JOB_TRIGGER
			building_icon = building_commercial_megaplex
			icon = merchant
		}
	}

	tags = { trade amenities }

	possible_pre_triggers = {
		has_owner = yes
		is_being_purged = no
		is_being_assimilated = no
		is_sapient = yes
	}
	possible_precalc = can_fill_ruler_job
	possible = {
		OR = {
			is_enslaved = no
			owner = { has_valid_civic = civic_du_contrat_social }
			owner = { has_valid_civic = civic_flower_of_evil }
		}
	}
	resources = {
		category = planet_merchants
		produces = {
			trade = 100
		}
		produces = {
			trigger = {
				exists = owner
				owner = {
					has_civic = civic_corporate_dominion
				}
			}
			unity = 10
		}
		produces = {
			trigger = {
				exists = owner
				owner = {
					has_tradition = tr_mercantile_marketplace_of_better_ideas
				}
			}
			unity = 10
		}

		produces = {
			trigger = {
				exists = owner
				owner = {
					OR = {
						has_valid_civic = civic_industrial_automatation
						has_valid_civic = civic_industrial_automatation_corporate
					}
				}
			}
			physics_research = 5
			engineering_research = 5
			society_research = 5
		}
		produces = {
			trigger = {
				exists = planet
				planet = {
					OR = {
						politician_is_merchant = yes
						politician_is_augmentation_merchant = yes
						politician_is_ashbroker = yes
					}
				}
			}
			unity = -15
			trade = 60
		}
		upkeep = {
			consumer_goods = 20
		}
	}
	planet_modifier = {
		trade_value_reduction = 0.5
	}
	triggered_planet_pop_group_modifier_for_all = {
		potential = {
			exists = owner
			owner = {
				has_policy_flag = market_operation
			}
		}
		modifier = {
			pop_cat_ruler_political_power = 0.5
		}
	}
	triggered_planet_pop_group_modifier_for_all = {
		potential = {
			exists = owner
			owner = {
				has_valid_civic = civic_triple_the_profit
			}
		}
		modifier = {
			pop_cat_ruler_political_power = 0.3
		}
	}
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				is_real_estate_market_country = yes
			}
		}
		modifier = {
			pop_housing_usage_mult = 0.5
			pop_category_specialists_energy_produces_add = 1.5
			pop_category_workers_energy_produces_add = 0.5
		}
	}
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				has_tradition = tr_mercantile_marketplace_of_better_ideas
			}
		}
		modifier = {
			planet_jobs_trade_produces_mult = 0.2
		}
	}
	triggered_planet_modifier = {
		potential = { always = yes }
		planet_amenities_add = 5000
		mult = value:scripted_modifier_mult|MODIFIER|pop_job_amenities_mult|
	}

	triggered_planet_modifier = {
		potential = {
			owner? = { has_valid_civic = civic_cultural_left }
		}
		planet_civilians_trade_produces_add = 10
	}

	weight = {
		weight = @ruler_job_weight
		mult = value:scripted_modifier_job_weight_mult|MODIFIER|pop_job_amenities_mult|
		# crisis purge
		modifier = {
			factor = 0.01
			exists = planet
			exists = planet.controller
			planet.controller = {
				OR = {
					is_country_type = swarm
					is_country_type = ai_empire
				}
			}
			OR = { # contingency won't purge itself
				NOT = { exists = event_target:custodian_bot }
				AND = {
					exists = event_target:custodian_bot
					NOT = { species = { is_same_value = event_target:custodian_bot } }
				}
			}
		}
	}

	inline_script = "jobs/automodding_priority_trade"
}
